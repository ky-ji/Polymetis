cmake_minimum_required(VERSION 3.16)
project(polymetis_joint_servo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(RUCKIG_SOURCE_DIR "" CACHE PATH "Verified Ruckig v0.15.3 source directory")
if(NOT EXISTS "${RUCKIG_SOURCE_DIR}/include/ruckig/ruckig.hpp")
  message(FATAL_ERROR "Provide the pinned Ruckig source; this build never fetches code")
endif()
option(BUILD_TORCH_SERVO "Build the TorchScript reference class" OFF)
if(BUILD_TORCH_SERVO)
  find_package(Torch REQUIRED)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
endif()
set(BUILD_CLOUD_CLIENT OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${RUCKIG_SOURCE_DIR}" ruckig)
add_executable(reference_test reference_test.cpp)
target_link_libraries(reference_test PRIVATE ruckig)
enable_testing()
add_executable(streaming_probe streaming_probe.cpp)
target_link_libraries(streaming_probe PRIVATE ruckig)
add_test(NAME streaming_probe COMMAND streaming_probe)
add_test(NAME streaming_probe_faster COMMAND streaming_probe "0.5")
add_test(NAME reference_test COMMAND reference_test)
add_executable(reference_replay_test reference_replay_test.cpp)
target_link_libraries(reference_replay_test PRIVATE ruckig)
add_test(NAME reference_replay_test COMMAND reference_replay_test
  "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/boundary_reversal.txt")
add_test(NAME predicted_reference_replay_test COMMAND reference_replay_test
  "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/boundary_reversal.txt" "0.04")
if(BUILD_TORCH_SERVO)
  add_library(polymetis_joint_servo SHARED torch_reference.cpp)
  target_link_libraries(polymetis_joint_servo PRIVATE ruckig "${TORCH_LIBRARIES}")
  install(TARGETS polymetis_joint_servo LIBRARY DESTINATION lib)
endif()
